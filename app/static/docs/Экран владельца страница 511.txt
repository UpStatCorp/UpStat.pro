    let currentMode = 'money';


    function renderMode(mode) {
      currentMode = mode;
      const data = dashboardData[mode];
      mainLabel.textContent = data.label;
      mainTitle.textContent = data.title;
      mainCopy.textContent = data.copy;
      mainContext.textContent = data.context;
      aiSee.textContent = data.aiSee;
      aiWhy.textContent = data.aiWhy;
      aiBecause.textContent = data.aiBecause;
      sharePreview.textContent = data.share;
      primaryAction.textContent = data.primaryText;
      secondaryAction.textContent = data.secondaryText;


      rankList.innerHTML = data.items.map((item, index) => `
        <div class="rank-item">
          <div class="rank-num">${index + 1}</div>
          <div>
            <div class="name">${item.name}</div>
            <div class="reason">${item.reason}</div>
          </div>
          <div class="rank-loss">
            <div class="value">${item.value}</div>
            <div class="tag">${item.tag}</div>
          </div>
        </div>
      `).join('');


      vizGrid.innerHTML = data.viz.map(card => `
        <div class="viz-card">
          <div class="viz-title">${card.title}</div>
          ${card.ring ? `
            <div class="ring-wrap">
              <div class="ring" data-value="${card.ring}"></div>
              <div>
                <div class="viz-value">${card.value}</div>
                <div class="viz-sub">${card.sub}</div>
              </div>
            </div>
          ` : `
            <div class="viz-value">${card.value}</div>
            <div class="viz-sub">${card.sub}</div>
          `}
        </div>
      `).join('');


      causeGrid.innerHTML = data.causes.map(cause => `
        <div class="cause-card">
          <div class="cause-head">
            <div>
              <div class="cause-title">${cause.title}</div>
              <div class="cause-desc">${cause.desc}</div>
            </div>
            <div class="cause-impact">${cause.impact}</div>
          </div>
        </div>
      `).join('');


      aiRecommendations.innerHTML = data.recommendations.map(item => `
        <div class="recommendation">
          <div class="title">${item.title}</div>
          <div class="desc">${item.desc}</div>
          <div class="roi">${item.roi}</div>
        </div>
      `).join('');


      modeButtons.forEach(btn => btn.classList.toggle('active', btn.dataset.mode === mode));
    }


    function showToast(title, body) {
      toast.innerHTML = `<strong>${title}</strong>${body}`;
      toast.classList.add('show');
      clearTimeout(showToast._t);
      showToast._t = setTimeout(() => toast.classList.remove('show'), 3200);
    }


    modeButtons.forEach(btn => {
      btn.addEventListener('click', () => renderMode(btn.dataset.mode));
    });
